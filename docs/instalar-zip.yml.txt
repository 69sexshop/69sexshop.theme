name: Instalar tema desde un zip

on:
  push:
    paths:
      - '*.zip'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  instalar:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Descomprimir los zips de la raíz
        run: |
          shopt -s nullglob
          zips=(*.zip)
          if [ ${#zips[@]} -eq 0 ]; then
            echo "No hay zips para instalar"
            exit 0
          fi
          for z in "${zips[@]}"; do
            unzip -o "$z" -x ".github/*" "__MACOSX/*"
            rm "$z"
          done
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add -A
          if git diff --cached --quiet; then
            echo "Sin cambios"
          else
            git commit -m "Instalar tema desde zip"
            git push origin HEAD
          fi
